Đồng bộ Cloud (Supabase) & Sao lưu

Dự án Supabase ielts-with-hani đã được gắn sẵn. Bạn chỉ cần nhập cùng một Mã đồng bộ (Sync ID) bí mật trên các thiết bị để đồng bộ dữ liệu.

Thông tin kết nối Supabase (đã gắn sẵn — bấm để xem / đổi)

SQL khởi tạo bảng trên Supabase (chạy 1 lần nếu chưa tạo bảng)

Mở SQL Editor của project ielts-with-hani → dán đoạn SQL dưới đây và bấm Run:

create table if not exists public.hani_profiles (
  id text primary key check (char_length(id) >= 4 and char_length(id) <= 128),
  data jsonb not null,
  updated_at timestamptz not null default now()
);
alter table public.hani_profiles enable row level security;
grant usage on schema public to anon, authenticated;
grant select, insert, update on table public.hani_profiles to anon, authenticated;

create policy "hani_profiles_select_by_sync_header"
  on public.hani_profiles for select to anon, authenticated
  using (('hani-sync:' || id) = coalesce(nullif(current_setting('request.headers', true), '')::jsonb ->> 'x-client-info', ''));

create policy "hani_profiles_insert_by_sync_header"
  on public.hani_profiles for insert to anon, authenticated
  with check (('hani-sync:' || id) = coalesce(nullif(current_setting('request.headers', true), '')::jsonb ->> 'x-client-info', ''));

create policy "hani_profiles_update_by_sync_header"
  on public.hani_profiles for update to anon, authenticated
  using (('hani-sync:' || id) = coalesce(nullif(current_setting('request.headers', true), '')::jsonb ->> 'x-client-info', ''))
  with check (('hani-sync:' || id) = coalesce(nullif(current_setting('request.headers', true), '')::jsonb ->> 'x-client-info', ''));
